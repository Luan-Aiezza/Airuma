using Godot;

/// <summary>
/// The attacks of the river of stars (stage 6, Mama Killa, the Moon, the goddess with four arms): the last boss, and the one that fills the screen. Six attacks in her calm form and four in her fury, each of five,
/// six or seven patterns at once. Her bullets are the light of the Moon: silver and violet, stars, crescents that fly as one, half moons, great white moons that orbit her and burst, galaxies that turn round her,
/// tides that cross the field from the sides, meteors from the sky, and, in her fury, the colour of blood. She stands on the horizon of the water (she goes to the sides and down, never above it) and fires from her
/// four hands. The figures are kept clear: every attack has one that is drawn on the screen as a whole, and the others fill the space between its lines.
/// </summary>
public static partial class Attacks
{
    // The four hands, from the outer pair to the inner pair.
    private static Vector2 OuterLeft(PatternContext c) => c.Origin + new Vector2(-72, -6);
    private static Vector2 OuterRight(PatternContext c) => c.Origin + new Vector2(72, -6);
    private static Vector2 InnerLeft(PatternContext c) => c.Origin + new Vector2(-34, 22);
    private static Vector2 InnerRight(PatternContext c) => c.Origin + new Vector2(34, 22);

    private static Shot Moonlight(float speed) => new Shot(BulletStyle.BeadWhite, speed).Sized(0.9f);
    private static Shot Silver(float speed) => new Shot(BulletStyle.OrbWhite, speed).Sized(0.95f);
    private static Shot BigMoon(float speed) => new Shot(BulletStyle.GiantWhite, speed).Hitting(34);

    /// <summary>The crescent moon: three crescents fly together at the boat, four arms of a galaxy turn from her, star dust falls from the whole sky, kunai from her outer hands, and hexagons of violet are laid out and let go.</summary>
    public static BossPhase KillaCrescent() => Attack("AttackKillaCrescent", BossMovementType.Roam, Centre, patterns: new[]
    {
        Pat.Every(2.0f, c => { for (int k = -1; k <= 1; k++) Pat.Crescent(c, c.Origin, c.Aim + k * 30, 64, 9, Moonlight(165)); }),
        Pat.Every(0.26f, c => Pat.PolarRing(c, c.Origin, 4, c.Count * 9f, 16, 82, 38, new Shot(BulletStyle.BeadCyan, 0).Sized(0.9f), scale: false)),
        Pat.Every(0.1f, c => Pat.Rain(c, Edge.Top, 3, 10, new Shot(BulletStyle.StarBlue, 115).Sized(0.9f))),
        Pat.Every(1.2f, c => Pat.Aimed(c, OuterLeft(c), 5, 36, new Shot(BulletStyle.KunaiBlue, 235))).After(0.3f),
        Pat.Every(1.2f, c => Pat.Aimed(c, OuterRight(c), 5, 36, new Shot(BulletStyle.KunaiBlue, 235))).After(0.9f),
        Pat.Every(3.2f, c => Pat.Polygon(c, c.Origin, 6, 78, c.Count * 15f, 4, new Shot(BulletStyle.OrbViolet, 1).LaunchedOut(0.7f, 150))).After(1.5f),
    })
        .With(1.40f,
            SideRain(0.1f, 2, BulletStyle.StarBlue, 140))
        .Plus(TopRain(0.1f, 2, BulletStyle.StarGold, 140),
            SideRain(0.12f, 2, BulletStyle.StarBlue, 160),
            Pat.Every(1.7f, c => Pat.Aimed(c, OuterLeft(c), 7, 50, new Shot(BulletStyle.KunaiViolet, 245))).After(0.6f).NotOnEasy());

    /// <summary>The tide: waves of orbs cross the field from both sides at once, each with a gap that rolls up and down; a rain of silver beads, rings of moonlight, a great blue moon at the boat that bursts, and a galaxy turning the other way.</summary>
    public static BossPhase KillaTide() => Attack("AttackKillaTide", BossMovementType.Roam, Centre, patterns: new[]
    {
        Pat.Every(0.8f, c => Pat.Wall(c, Edge.Left, 32, 0.74f + 0.14f * Pat.Wave(c.Time, 6), 150, new Shot(BulletStyle.OrbBlue, 125).Sized(0.9f))),
        Pat.Every(0.8f, c => Pat.Wall(c, Edge.Right, 32, 0.74f - 0.14f * Pat.Wave(c.Time, 6), 150, new Shot(BulletStyle.OrbCyan, 125).Sized(0.9f))).After(0.4f),
        Pat.Every(0.12f, c => Pat.Rain(c, Edge.Top, 3, 8, new Shot(BulletStyle.BeadCyan, 130).Sized(0.8f))),
        Pat.Every(1.0f, c => Pat.Ring(c, c.Origin, 28, c.Count * 6f, Silver(110))),
        Pat.Every(2.8f, c => c.Fire(c.Origin, c.Aim, BigMoon(110).Styled(BulletStyle.GiantBlue).Splitting(1.5f, 22, BulletStyle.OrbBlue, 130))).After(1),
        Pat.Every(0.3f, c => Pat.PolarRing(c, c.Origin, 3, c.Count * -11f, 16, 78, -34, new Shot(BulletStyle.BeadViolet, 0).Sized(0.9f), scale: false)).After(1),
    })
        .Pressured(1.40f)
        .Plus(TopRain(0.1f, 2, BulletStyle.StarGold, 140),
            SideRain(0.12f, 2, BulletStyle.StarBlue, 160),
            Pat.Every(1.7f, c => Pat.Aimed(c, OuterLeft(c), 7, 50, new Shot(BulletStyle.KunaiViolet, 245))).After(0.6f).NotOnEasy())
        .Pressured(1.08f);

    /// <summary>The half moon: concentric half circles of silver open from her, one inside the other, over roses of five petals and windmills; fine lattices of beads fall slowly, and pentagrams of stars are laid out and let go.</summary>
    public static BossPhase KillaHalf() => Attack("AttackKillaHalf", BossMovementType.Roam, Centre, patterns: new[]
    {
        Pat.Every(1.1f, c =>
        {
            float aim = c.Count % 2 == 0 ? 90 : c.Aim;
            foreach (var (speed, n) in new[] { (110f, 21), (150f, 25), (190f, 29) }) Pat.Arc(c, c.Origin, n, aim, 180, Silver(speed), scale: false);
        }),
        Pat.Every(0.9f, c => Pat.Rose(c, c.Origin, 5, 40, c.Count * 7f, 0.6f, new Shot(BulletStyle.OrbViolet, 105).Sized(0.9f))).After(0.4f),
        Pat.Every(2.0f, c => Pat.Spokes(c, InnerLeft(c), 5, 4, c.Count * 13f, 22, new Shot(BulletStyle.BeadCyan, 90).Sized(0.9f))).After(0.5f),
        Pat.Every(2.0f, c => Pat.Spokes(c, InnerRight(c), 5, 4, c.Count * -13f, 22, new Shot(BulletStyle.BeadPink, 90).Sized(0.9f))).After(1.5f),
        Pat.Every(0.9f, c => Pat.Lattice(c, 60, c.Count * 7f, new Shot(BulletStyle.BeadWhite, 70).Sized(0.8f), new Shot(BulletStyle.BeadWhite, 70).Sized(0.8f), c.MiddleX + 160 * Pat.Wave(c.Time, 9), 520, 120)).After(1),
        Pat.Every(3.4f, c => Pat.Star(c, c.Origin, 5, 2, 95, c.Count * 17f, 5, new Shot(BulletStyle.StarGold, 1).LaunchedOut(0.8f, 150))).After(2),
    })
        .Pressured(1.47f)
        .Plus(TopRain(0.1f, 2, BulletStyle.StarGold, 140),
            SideRain(0.12f, 2, BulletStyle.StarBlue, 160),
            Pat.Every(1.7f, c => Pat.Aimed(c, OuterLeft(c), 7, 50, new Shot(BulletStyle.KunaiViolet, 245))).After(0.6f).NotOnEasy())
        .Pressured(1.00f);

    /// <summary>The full moon: three great moons orbit her and fire rings of their own as they go, a rose turns from her centre, rain of stars, combs of kunai from the sides, and fans from her four hands in turn.</summary>

        Pat.Every(1.6f, c => Pat.Aimed(c, (c.Count % 4) switch { 0 => OuterLeft(c), 1 => InnerLeft(c), 2 => InnerRight(c), _ => OuterRight(c) }, 7, 50, new Shot(BulletStyle.OrbCyan, 225).Sized(0.9f))).After(0.5f),
        Pat.Every(4.5f, c => c.Fire(c.Origin, 90, BigMoon(100).Splitting(1.6f, 26, BulletStyle.BeadWhite, 120))).After(2).NotOnEasy(),
    })
        .Pressured(1.23f)
        .Plus(TopRain(0.1f, 2, BulletStyle.StarGold, 140),
            SideRain(0.12f, 2, BulletStyle.StarBlue, 160),
            Pat.Every(1.7f, c => Pat.Aimed(c, OuterLeft(c), 7, 50, new Shot(BulletStyle.KunaiViolet, 245))).After(0.6f).NotOnEasy())
        .Pressured(1.00f);
